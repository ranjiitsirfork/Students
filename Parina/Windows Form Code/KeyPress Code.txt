private void AllowOnlyText(object sender, KeyPressEventArgs e)
        {
            if (! (char.IsLetter(e.KeyChar) || (e.KeyChar == (char)Keys.Back) || (char.IsControl(e.KeyChar)) || (char.IsWhiteSpace(e.KeyChar))) )
            {
                e.Handled = true;
            }
        }

        private void AllowOnlyNumbers(object sender, KeyPressEventArgs e)
        {
            if (!(char.IsDigit(e.KeyChar) || (e.KeyChar == (char)Keys.Back) || (char.IsControl(e.KeyChar))) )
            {
                e.Handled = true;
            }
        }

        private void AllowOnlyTextNumbers(object sender, KeyPressEventArgs e)
        {
            if (!(char.IsLetterOrDigit(e.KeyChar) || (e.KeyChar == (char)Keys.Back) || (char.IsControl(e.KeyChar)) || (char.IsWhiteSpace(e.KeyChar))) )
            {
                e.Handled = true;
            }
        }
